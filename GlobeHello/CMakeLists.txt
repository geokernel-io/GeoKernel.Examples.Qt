cmake_minimum_required(VERSION 3.23)
project(GlobeHello LANGUAGES CXX)
include(GlobeSample.cmake)
find_package(Qt6 6.11.0 EXACT REQUIRED COMPONENTS Widgets)
geokernel_add_globe_sample(GlobeHello main.cpp resources/icons.qrc)
set_target_properties(GlobeHello PROPERTIES AUTORCC ON)
if(WIN32)
    enable_language(RC)
    target_sources(GlobeHello PRIVATE resources/GlobeHello.rc)
endif()
set_target_properties(GlobeHello PROPERTIES WIN32_EXECUTABLE TRUE)
target_link_libraries(GlobeHello PRIVATE Qt6::Widgets)
